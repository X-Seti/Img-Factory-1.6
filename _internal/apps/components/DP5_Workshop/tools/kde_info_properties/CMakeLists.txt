# X-Seti - October 27 2026 - IMG Factory 1.6 - Amiga .info KFileMetaData extractor (Dolphin Properties)
cmake_minimum_required(VERSION 3.16)
project(amigainfoextractor CXX)
set(KF_MAJOR 6 CACHE STRING "KDE Frameworks major version (6 for Plasma 6)")
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs)
include(KDECMakeSettings)
find_package(Qt${KF_MAJOR} REQUIRED COMPONENTS Core)
find_package(KF${KF_MAJOR}FileMetaData REQUIRED)
add_library(amigainfoextractor MODULE amigainfoextractor.cpp)
target_link_libraries(amigainfoextractor Qt${KF_MAJOR}::Core KF${KF_MAJOR}::FileMetaData)
install(TARGETS amigainfoextractor DESTINATION ${KDE_INSTALL_PLUGINDIR}/kf${KF_MAJOR}/kfilemetadata)
